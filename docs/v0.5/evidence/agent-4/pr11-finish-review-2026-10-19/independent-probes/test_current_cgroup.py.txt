"""Independent current-capability test. Ordinary files; no Linux operations."""
import ast
import concurrent.futures
import inspect
import os
import threading
from pathlib import Path
import pytest
from output_guardian.model import Closed
from output_guardian.v2 import linux_fence as lf


def leaf(tmp_path, monkeypatch):
    p=tmp_path/'leaf';p.mkdir(mode=0o700)
    for name,value in [('cgroup.procs',''),('cgroup.kill',''),('cgroup.events','populated 0\n')]:
        (p/name).write_text(value);(p/name).chmod(0o600)
    g=lf.OwnedCgroup.__new__(lf.OwnedCgroup)
    g.path,g.inode,g.device=p,p.stat().st_ino,p.stat().st_dev
    g.members,g.retired,g._lock={},False,threading.RLock()
    monkeypatch.setattr(lf,'_cgroup2',lambda p:None)
    return g


@pytest.mark.parametrize('operation',['fence','close','attach'])
def test_same_empty_leaf_never_authorizes_mutation(tmp_path,monkeypatch,operation):
    g=leaf(tmp_path,monkeypatch)
    # A native supervisor identity itself is not a child; no body flag can help.
    args=(lf.identity(os.getpid()),) if operation=='attach' else ()
    for name in ['exclusive','controller','verified','allow_mutation']:setattr(g,name,True)
    with pytest.raises(Closed):getattr(g,operation)(*args)
    assert (g.path/'cgroup.kill').read_bytes()==b''
    assert (g.path/'cgroup.procs').read_bytes()==b''
    assert g.path.is_dir() and g.retired


def test_no_mutating_syscall_or_positive_receipt_in_scope_class():
    tree=ast.parse(inspect.getsource(lf.OwnedCgroup))
    forbidden={'write','write_text','write_bytes','mkdir','rmdir','unlink','remove','kill','system','Popen'}
    for n in ast.walk(tree):
        if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute):assert n.func.attr not in forbidden
    # Current methods have no return expression containing a purported receipt.
    assert not any(isinstance(n,ast.Return) and isinstance(n.value,ast.Dict) for n in ast.walk(tree))


def test_concurrent_repeated_retired_handle_stays_denied(tmp_path,monkeypatch):
    g=leaf(tmp_path,monkeypatch)
    def attempt(i):
        try:getattr(g,'fence' if i%2 else 'close')()
        except Closed:return 'DENIED'
        return 'UNEXPECTED_POSITIVE'
    with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:assert set(pool.map(attempt,range(40)))=={'DENIED'}
    assert (g.path/'cgroup.kill').read_bytes()==b'' and (g.path/'cgroup.procs').read_bytes()==b''
    assert g.path.is_dir()


def test_snapshot_changes_after_last_check_cannot_authorize(tmp_path,monkeypatch):
    g=leaf(tmp_path,monkeypatch);deny=g._exclusive_mutation
    def late():
        (g.path/'late').mkdir();(g.path/'late'/'cgroup.procs').write_text('999999\n')
        deny()
    monkeypatch.setattr(g,'_exclusive_mutation',late)
    with pytest.raises(Closed):g.fence()
    assert (g.path/'cgroup.kill').read_bytes()==b''


def test_missing_lock_and_caught_failure_do_not_restore_authority(tmp_path,monkeypatch):
    g=leaf(tmp_path,monkeypatch);del g._lock
    for _ in range(3):
        with pytest.raises(Closed):g.fence()
        # Simulate an untrusted caller resetting local retired flag.
        g.retired=False
    assert (g.path/'cgroup.kill').read_bytes()==b''


def test_constructor_no_allocation_even_with_matching_parent(tmp_path,monkeypatch):
    (tmp_path/'cgroup.controllers').write_text('cpu');tmp_path.chmod(0o700)
    monkeypatch.setattr(lf.sys,'platform','linux');monkeypatch.setattr(lf,'_cgroup2',lambda p:None)
    before=set(tmp_path.iterdir())
    with pytest.raises(Closed,match='OUTPUT_FENCE_UNPROVEN'):lf.OwnedCgroup(tmp_path)
    assert set(tmp_path.iterdir())==before
