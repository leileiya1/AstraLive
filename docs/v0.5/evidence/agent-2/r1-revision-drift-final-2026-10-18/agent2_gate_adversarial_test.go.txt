	t.Helper()

func currentGate(f *interop) *InstallationGate { return reviewRecord(f).gate }
func closeServer(t *testing.T, f *interop) {
	t.Helper()
	if e := f.server.Close(); e != nil {
		t.Fatal(e)
	}
}
func latchState(t *testing.T, f *interop) (int, string) {
	t.Helper()
	var n int
	var s string
	if e := f.s.DB.QueryRow(f.ctx, "SELECT count(*) FROM dl05_receiver_consumptions WHERE command_id=$1", f.cmd["commandId"]).Scan(&n); e != nil {
		t.Fatal(e)
	}
	if n > 0 {
		if e := f.s.DB.QueryRow(f.ctx, "SELECT state FROM dl05_v2_execution_state WHERE command_id=$1", f.cmd["commandId"]).Scan(&s); e != nil {
			t.Fatal(e)
		}
	}
	return n, s
}
func waitFile(t *testing.T, p string) {
	t.Helper()
	end := time.Now().Add(2 * time.Second)
	for {
		if _, e := os.Stat(p); e == nil {
			return
		}
		if time.Now().After(end) {
			t.Fatal("fixture marker missing")
		}
		time.Sleep(time.Millisecond)
	}
}

func TestAgent2FinalCASConflictRequiresDurableRevoke(t *testing.T) {
	for _, kind := range []string{"persisted-revoked", "not-revoked"} {
		t.Run(kind, func(t *testing.T) {
			f, _ := newLifecycle(t)
			f.ready()
			if v := f.remote("CommitReceiverConsume", f.consumeQ); v["ok"] != true {
				t.Fatal(v)
			}
			closeServer(t, f)
			original := f.s.Validate
			var triggered atomic.Bool
			f.s.Validate = func(name string, d authority.Document) error {
				if e := original(name, d); e != nil {
					return e
				}
				if name == "RevokeReceiverBootstrapRequest" && triggered.CompareAndSwap(false, true) {
					if kind == "not-revoked" {
						return &authority.Error{Code: "STATE_CONFLICT"}
					}
					op, _ := wire.LookupOperation(wire.OpRevokeReceiverBootstrap)
					wc := wire.RequestContext{Role: wire.RoleControlRegistrar, Target: op.Target, Method: op.Method, RouteTemplate: op.Path}
					_, e := f.s.Handle(f.ctx, f.registrar, op.ID, wc, mustJSON(d))
					return e
				}
				return nil
			}
			e := f.registry.RetireInstallation(f.ctx, f.registrar, string(f.binding.CommandID))
			if !triggered.Load() {
				t.Fatal("CAS seam not exercised")
			}
			if kind == "not-revoked" {
				if e == nil || reviewRecord(f).settled.Load() || currentGate(f).valid() {
					t.Fatal("CAS conflict treated as durable revoke", e)
				}
				f.s.Validate = original
				if e = f.registry.RetireInstallation(f.ctx, f.registrar, string(f.binding.CommandID)); e != nil {
					t.Fatal(e)
				}
			} else if e != nil {
				t.Fatal(e)
			}
			assertRetired(t, f)
			if n, s := latchState(t, f); n != 1 || s != "UNKNOWN" {
				t.Fatal(n, s)
			}
		})
	}
}

func TestAgent2FinalOwnedPostgresOutageDuringRetire(t *testing.T) {
	f, source := newLifecycle(t)
	f.ready()
	if v := f.remote("CommitReceiverConsume", f.consumeQ); v["ok"] != true {
		t.Fatal(v)
	}
	closeServer(t, f)
	root := os.Getenv("AGENT4_RECEIVER1_OWNED_ROOT")
	b, e := os.ReadFile(filepath.Join(root, "OWNED_AGENT2"))
	if e != nil || string(b) != "08af1049250d2657d7afb5e303d97a02e9761570" || !strings.HasPrefix(root, "/tmp/a2-r1-final-") {
		t.Fatal("refuse non-owned PG")
	}
	port := os.Getenv("AGENT2_R1_PG_PORT")
	if _, e = strconv.Atoi(port); e != nil {
		t.Fatal(e)
	}
	ctl := "/opt/homebrew/opt/postgresql@17/bin/pg_ctl"
	run := func(args ...string) {
		t.Helper()
		if out, e := exec.Command(ctl, args...).CombinedOutput(); e != nil {
			t.Fatalf("owned PG operation %v: %s", e, out)
		}
	}
	source.gate.CloseAdmission()
	run("-D", filepath.Join(root, "data"), "-m", "fast", "-w", "stop")
	restart := func() {
		run("-D", filepath.Join(root, "data"), "-l", filepath.Join(root, "evidence/postgres.txt"), "-o", "-h 127.0.0.1 -p "+port+" -k "+filepath.Join(root, "sockets"), "-w", "start")
	}
	stopped := true
	defer func() {
		if stopped {
			restart()
		}
	}()
	ctx, cancel := context.WithTimeout(f.ctx, 200*time.Millisecond)
	e = f.registry.RetireInstallation(ctx, f.registrar, string(f.binding.CommandID))
	cancel()
	if e == nil || source.gate.valid() || source.revision.Load() != 1 {
		t.Fatal("DB failure allowed replacement", e)
	}
	restart()
	stopped = false
	// Restart can expose a previously idle pooled connection's 57P01 once.
	// Retirement retries are bounded cleanup only; every error must remain closed
	// and never permit a physical replacement or a new positive Consume.
	for attempt := 0; attempt < 3; attempt++ {
		e = f.registry.RetireInstallation(f.ctx, f.registrar, string(f.binding.CommandID))
		if e == nil {
			break
		}
		var pe *pgconn.PgError
		if !errors.As(e, &pe) || pe.Code != "57P01" {
			t.Fatal("unexpected post-restart retirement failure", e)
		}
		if source.gate.valid() || source.revision.Load() != 1 {
			t.Fatal("stale pooled connection reopened source")
		}
		t.Logf("explicit retirement retry %d: terminated idle PG connection 57P01, admission remains closed", attempt+1)
	}
	if e != nil {
		t.Fatal("bounded retirement recovery failed", e)
	}
	assertRetired(t, f)
	if n, s := latchState(t, f); n != 1 || s != "UNKNOWN" {
		t.Fatal(n, s)
	}
}

func TestAgent2FinalResponseConstructionReaderDrain(t *testing.T) {
	f, source := newLifecycle(t)
	f.ready()
	closeServer(t, f)
	arrived, release := make(chan struct{}), make(chan struct{})
	srv, e := NewServer(f.s, f.registry, f.g.managed.Pin)
	if e != nil {
		t.Fatal(e)
	}
	op, _ := wire.LookupOperation(wire.OpCommitReceiverConsume)
	srv.http.Handler = http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path == op.Path {
			w = &reviewWriter{ResponseWriter: w, at: arrived, release: release}
		}
		srv.ServeHTTP(w, r)
	})
	f.server = srv
	if e = srv.Listen(filepath.Join(f.root, "authority.sock")); e != nil {
		t.Fatal(e)
	}
	gotCh := make(chan authority.Document, 1)
	go func() { gotCh <- f.remote("CommitReceiverConsume", f.consumeQ) }()
	<-arrived
	retired := make(chan error, 1)
	go func() { retired <- f.registry.RetireInstallation(f.ctx, f.registrar, string(f.binding.CommandID)) }()
	waitGateClosed(t, source.gate)
	select {
	case <-retired:
		t.Fatal("reader released before response write finished")
	default:
	}
	if source.revision.Load() != 1 {
		t.Fatal("physical changed")
	}
	close(release)
	got := <-gotCh
	if e = <-retired; e != nil {
		t.Fatal(e)
	}
	assertRetired(t, f)
	// Already committed responses can be in flight. Do not confuse a network ACK
	// delivered after close with a NEW post-retirement right; final Receiver proof
	// and physical sink fencing are independent. Record actual behavior exactly.
	t.Logf("pre-close SQL committed ACK delivered after CloseAdmission: rpcOK=%v true=%v; physical tuple unchanged until reader drain", got["ok"], testObj(got, "result")["consumeAuthorized"])
	if n, s := latchState(t, f); n != 1 || s != "UNKNOWN" {
		t.Fatal(n, s)
	}
}

type reviewWriter struct {
	http.ResponseWriter
	at      chan struct{}
	release chan struct{}
}

func (w *reviewWriter) WriteHeader(n int) {
	if n == 200 {
		close(w.at)
		<-w.release
	}
	w.ResponseWriter.WriteHeader(n)
}

	cmd.Env = append(os.Environ(), "AGENT2_R1_CRASH_CHILD=owned")
	pipe, e := cmd.StdoutPipe()
	if e != nil {
		t.Fatal(e)
	}
	if e = cmd.Start(); e != nil {
		t.Fatal(e)
	}
	defer func() { cmd.Process.Kill(); cmd.Wait() }()
	lines := bufio.NewScanner(pipe)
	var data struct {
		Command   authority.Document            `json:"command"`
		Binding   wire.ReceiverBootstrapBinding `json:"binding"`
		Control   authority.Actor               `json:"control"`
		Registrar authority.Actor               `json:"registrar"`
		Runtime   Identity                      `json:"runtime"`
		Second    Identity                      `json:"second"`
		Guardian  Identity                      `json:"guardian"`
	}
	seen := false
	for lines.Scan() {
		if strings.HasPrefix(lines.Text(), "A2_OWNED_RETIRE_CLOSED ") {
			if e = json.Unmarshal([]byte(strings.TrimPrefix(lines.Text(), "A2_OWNED_RETIRE_CLOSED ")), &data); e != nil {
				t.Fatal(e)
			}
			seen = true
			break
		}
	}
	if !seen {
		t.Fatal("owned child did not reach gate-closed retirement")
	}
	for _, id := range []Identity{data.Runtime, data.Second, data.Guardian} {
		id := id
		t.Cleanup(func() {
			if now, e := InspectProcess(id.PID); e == nil && now == id {
				p, e := os.FindProcess(id.PID)
				if e == nil {
					p.Kill()
				}
			}
		})
	}
	if e = cmd.Process.Kill(); e != nil {
		t.Fatal(e)
	}
	death := cmd.Wait()
	var killed *exec.ExitError
	if !errors.As(death, &killed) {
		t.Fatal("owned helper did not terminate with SIGKILL", death)
	}
	status, ok := killed.Sys().(syscall.WaitStatus)
	if !ok || !status.Signaled() || status.Signal() != syscall.SIGKILL {
		t.Fatal("wrong owned crash signal", death)
	}
	t.Logf("owned Go child PID=%d killed with SIGKILL after gate-closed marker; native children=%d,%d,%d checked by PID+birth before cleanup", cmd.Process.Pid, data.Runtime.PID, data.Second.PID, data.Guardian.PID)
	pool, e := pgxpool.New(ctx, os.Getenv("AGENT4_RECEIVER1_DATABASE_URL"))
	if e != nil {
		t.Fatal(e)
	}
	defer pool.Close()
	store := authority.New(pool)
	registry, e := NewRegistry(store)
	if e != nil {
		t.Fatal(e)
	}
	if _, _, e = registry.hold(ctx, data.Registrar, data.Binding); e == nil {
		t.Fatal("fresh root reconstructed old capability")
	}
	command := data.Command["commandId"]
	var n int
	if e = pool.QueryRow(ctx, "SELECT count(*) FROM dl05_receiver_consumptions WHERE command_id=$1", command).Scan(&n); e != nil || n != 1 {
		t.Fatal("crash lost latch", n, e)
	}
	// An authenticated test CONTROL marks the unresolved execution UNKNOWN using
	// the frozen operation; it does not recreate a source/ticket or replay it.
	q := authority.Document{"schemaVersion": "0.5", "contractRevision": "durable-live-2", "requestId": fresh("unknown_"), "tenantId": data.Control.Tenant, "ownerId": data.Control.Owner, "runId": data.Command["runId"], "expectedRevision": 1, "runEpoch": 1, "attemptId": data.Command["attemptId"], "lastKnownState": "DISPATCHED", "reasonCode": "ACK_LOST", "evidenceRef": "test:owned-backend-sigkill-during-retire", "idempotencyKey": "unknown:retire-crash"}
	op, _ := wire.LookupOperation(wire.OpMarkExecutionUnknown)
	wc := wire.RequestContext{Role: wire.RoleControl, Target: op.Target, Method: op.Method, RouteTemplate: op.Path}
	if _, e = store.Handle(ctx, data.Control, op.ID, wc, mustJSON(q)); e != nil {
		t.Fatal(e)
	}
	var state string
	pool.QueryRow(ctx, "SELECT state FROM dl05_v2_execution_state WHERE command_id=$1", command).Scan(&state)
	if state != "UNKNOWN" {
		t.Fatal(state)
	}
	t.Log("SIGKILL only owned Go child; fresh Registry denies old identity; latch retained and explicit CONTROL audit UNKNOWN")
}

// The coordinated capability does not replace equality of the COMPLETE tuple.
type finalTupleSource struct {
	physicalFixture
	field string
}

func (s finalTupleSource) InstallationGate(ctx context.Context, b wire.ReceiverBootstrapBinding, r, g Identity) (*InstallationGate, error) {
	if s.field == "nil-gate" {
		return nil, nil
	}
	v, e := s.physicalFixture.Observe(ctx, b, r, g)
	if e != nil {
		return nil, e
	}
	switch s.field {
	case "revision":
		v.InstallationRevision++
	case "session":
		v.Binding.SessionID = "foreign-session"
	case "generation":
		v.Binding.GenerationID++
	case "output-binding":
		v.Binding.OutputBindingVersion++
	case "runtime":
		v.Binding.RuntimeIncarnation = "old-runtime"
	case "guardian":
		v.GuardianPrincipalID = "old-guardian"
	case "observer":
		v.ObservedByPrincipalID = "foreign-registrar"
	case "fence":
		v.ClosedFenceProofRef = "foreign-fence"
	}
	return NewInstallationGate(v)
}
func TestAgent2FinalGatePinsFullImmutableTuple(t *testing.T) {
	f := newInterop(t)
	for _, field := range []string{"nil-gate", "revision", "session", "generation", "output-binding", "runtime", "guardian", "observer", "fence"} {
		t.Run(field, func(t *testing.T) {
			rt, g := f.launch("REGISTERED_RUNTIME"), f.launch("OUTPUT_GUARDIAN")
			_, e := f.registry.Install(f.ctx, f.registrar, rt.managed.Pin, g.managed.Pin, finalTupleSource{physicalFixture{f.registrar.ID, g.managed.Pin.actor.ID}, field})
			var ae *authority.Error
			if !errors.As(e, &ae) || ae.Code != "OUTPUT_FENCE_UNPROVEN" {
				t.Fatalf("tuple mismatch did not fail at physical trust check: %v", e)
			}
		})
	}
	if f.count("dl05_v2_installations") != 1 || f.count("dl05_receiver_bootstraps") != 0 {
		t.Fatal("bad capability mirrored or registered")
	}
}
